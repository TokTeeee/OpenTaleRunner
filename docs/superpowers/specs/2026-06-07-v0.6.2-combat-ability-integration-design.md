# v0.6.2 — Ability 统一 + 战斗接入 (Ability Unification + Combat Integration)

> 继承自：v0.6.1 (commit `3c74d2f`, **数据层已被 v0.6.2 重构**, 见 §1)
> 锁定日期：2026-06-07
> 父版本：v0.6 魔法/祷告 — B 子系统（战斗接入）+ v0.6.1 数据层重设。
> 后续：v0.6.3（物品 & affix 池） / v0.6.4（学习 UI & 叙事）。

---

## §0 背景

v0.6.1 spec 写了 Spell/Prayer **双类型**（6 元素魔法 + 2 元素祷告）。v0.6.2 brainstorm 时新增需求：
- **战技**（BattleArt）作为"物理/物理-buff/物理-治疗"类技能，**与法术/祷告机制相同**（伤害/BUFF/治疗三型），不应单独建类型
- 玩家期望 ActionMenu 用 1 个"技能"入口, 弹层中按 3 school (魔法/祷告/战技) 分类展示
- v0.6.1 的 Spell/Prayer 双类型已不再合适,需重构为统一 `Ability` 数据模型

**结论**: v0.6.2 = (a) 重设 v0.6.1 数据层为统一 Ability + (b) 战斗接入。本 spec 包含这两部分。

```typescript
export const ELEMENT_LABELS: Record<Element, string> = {

  magic: '魔法', prayer: '祷告', battle_art: '战技',
};

export const SCHOOL_ICONS: Record<AbilitySchool, string> = {
  magic: '✨', prayer: '🌟', battle_art: '⚔️',
};

export interface AbilityDescription {
  shortEffect: string;
  narrative: string;
  visualTag: string;
}

/** 战技 special 效果 (仅 battle_art 用) */
export type BattleArtSpecial =
  | 'high_crit'              // 命中时 +30% 暴击 (×1.3 伤害)
  | 'armor_pierce'           // 计算伤害时 target.defense 减半
  | 'life_steal'             // 30% 伤害转治疗
  | 'self_dodge_penalty';    // 自身 -1 招架门槛 1 回合

/** 单一效果模型 (覆盖原 Spell.damageFormula + Prayer.effect) */
export interface AbilityEffect {
  /** 伤害骰 (正数). isHeal=true 时, 值会被翻为治疗 */
  damageDice: string;          // '1d6' / '1d6+2' / '0' (纯 buff)
  isHeal: boolean;             // true = 治疗 (对 target 增 HP), false = 伤害
  /** 主要属性缩放: 伤害/治疗 += floor((attribute-10)/2) */
  attributeScale: AttributeName;  // 'STR' | 'DEX' | 'INT' | 'WIS' | 'CON' | 'CHA'
  /** 元素. 战技 = null (物理, 走 attack 同公式) */
  element: Element | null;
  /** buff (可选) */
  applyBuff?: { ref: string; stacks: number; turns: number };
  /** 战技 special (可选, 仅 battle_art) */
  special?: BattleArtSpecial;
}

export interface AbilityRequirement {
  /** 允许的职业, 'any' = 任意职业可学 */
  classes: Array<'warrior' | 'mage' | 'cleric' | 'thief' | 'any'>;
  /** 至少一项属性 ≥ 门槛即可 */
  minAttribute: Partial<Attributes>;  // e.g. { INT: 12 } or { STR: 13 }
  minLevel: number;
}

export interface Ability {
  id: string;            // 'spell_fire_bolt' / 'prayer_holy_heal' / 'art_warrior_smash'
  name: string;
  school: AbilitySchool;
  /** 元素. magic/prayer 必填, battle_art = null */
  element: Element | null;
  tier: AbilityTier;
  requirements: AbilityRequirement;
  cost: { ap: number; mp: number };  // mp 可为 0 (战技, 0 MP)
  target: AbilityTarget;
  effect: AbilityEffect;
  description: AbilityDescription;
}
```

### 2.5 学习校验（合并 — 替代 v0.6.1）

```typescript
// services/abilities/learnAbility.ts (重写)

export interface LearnCheckInput {
  character: Pick<Character, 'classId' | 'level' | 'attributes'>;
  ability: Ability;
}

export type LearnCheckResult =
  | { canLearn: true }
  | { canLearn: false; reason: 'class' | 'attribute' | 'level'; required: string };

export function checkCanLearn(input: LearnCheckInput): LearnCheckResult {
  const { character, ability } = input;
  const r = ability.requirements;

  // 1. 职业
  const classOk = r.classes.includes('any') || (character.classId !== null && r.classes.includes(character.classId as 'mage' | 'cleric' | 'warrior' | 'thief'));
  if (!classOk) {
    return { canLearn: false, reason: 'class', required: r.classes.join(' 或 ') };
  }
  // 2. 属性 (Partial<Attributes> 任一项 ≥ 门槛)
  for (const [attr, threshold] of Object.entries(r.minAttribute)) {
    const v = character.attributes[attr as AttributeName];
    if (v < (threshold as number)) {
      return { canLearn: false, reason: 'attribute', required: `${attr} ≥ ${threshold}` };
    }
  }
  // 3. 等级
  if (character.level < r.minLevel) {
    return { canLearn: false, reason: 'level', required: `等级 ≥ ${r.minLevel}` };
  }
  return { canLearn: true };
}
```

## §5 ActionResolver: resolveAbility

### 5.1 主流程

```typescript
// services/combat/ActionResolver.ts (扩展)

private resolveAbility(
  action: Extract<CombatAction, { kind: 'ability' }>,
  ctx: ResolverContext,
  state: CombatState,
): TurnResult {
  const user = state.combatants[action.userId];
  const target = action.targetId ? state.combatants[action.targetId] : null;
  const ability = getAbility(action.abilityId);
  const log: CombatLogEntry[] = [];

  // 1. 校验
  if (!ability) { log.push(errorLog('未知 ability')); return { log, buffTicks: [], ended: false }; }
  if (!user || user.isDead) { log.push(errorLog('无效 user')); return { log, buffTicks: [], ended: false }; }
  if (ability.cost.ap > user.ap) { log.push(errorLog('AP 不足')); return { log, buffTicks: [], ended: false }; }
  if (ability.cost.mp > (user.mp ?? 0)) { log.push(errorLog('MP 不足')); return { log, buffTicks: [], ended: false }; }
  if (target && target.isDead) { log.push(errorLog('目标已死')); return { log, buffTicks: [], ended: false }; }

  // 2. 扣 AP + MP
  useCombatStore.getState().applyAP(user.id, -ability.cost.ap);
  if (ability.cost.mp > 0) useCombatStore.getState().applyMP(user.id, -ability.cost.mp);

  // 3. 命中投 (d20 + DEX_mod, 与 attack 同公式)
  const qteRes = ctx.qte({ action, attacker: user, target: target!, state });
  const toHit = rollToHit(user, ctx.roll);
  const threshold = target ? hitThreshold(target, /* dodgePenalty */ 0, false) : { total: 0 };
  const hit = !target || checkHit(toHit.total, threshold.total);

  log.push({
    kind: 'action',
    round: state.round,
    turn: state.turn,
    message: `${user.name} 释放 ${SCHOOL_LABELS[ability.school]} · ${ability.name} → ${target?.name ?? '自身'}: d20=${toHit.d20}+${toHit.dexMod}=${toHit.total} vs ${threshold.total}`,
    data: { ability, toHit, threshold, qte: qteRes },
    timestamp: Date.now(),
  });

  // 4. 算 baseValue
  const diceResult = rollDice(ability.effect.damageDice, ctx.roll);
  const attrValue = effectiveAttribute(user, ability.effect.attributeScale);
  const attrMod = Math.floor((attrValue - 10) / 2);
  const baseValue = Math.max(0, diceResult.total + attrMod);

  // 5. 应用战技 special (currentTurn 用于 buff 字段)
  const specialResult = applySpecial(ability.effect.special, baseValue, user, target, state.turn);
  const damageAfterSpecial = specialResult.damage;

  // 6. 应用抗性 (only for magic/prayer, not battle_art)
  let finalValue = damageAfterSpecial;
  if (ability.element !== null && target) {
    finalValue = applyResistance(damageAfterSpecial, ability.element, target.elementalResistances);
  }

  // 7. 应用 QTE 缩放 (同 attack 公式)
  finalValue = Math.max(0, Math.round(finalValue * (1 + qteRes.modifier * this.damageScale)));

  // 8. isHeal 决定加血还是扣血
  if (ability.effect.isHeal) {
    useCombatStore.getState().applyHeal(target?.id ?? user.id, finalValue);
  } else if (target) {
    useCombatStore.getState().applyDamage(target.id, finalValue);
  }

  // 9. push buff (if any)
  if (ability.effect.applyBuff && target) {
    useCombatStore.getState().addBuff(target.id, {
      ref: ability.effect.applyBuff.ref,
      stacks: ability.effect.applyBuff.stacks,
      remainingTurns: ability.effect.applyBuff.turns,
      source: user.id,
      appliedAtTurn: state.turn,
      modifiers: {},
    });
  }

  // 10. 写 log
  log.push({
    kind: 'action',
    round: state.round,
    turn: state.turn,
    message: `${ability.effect.isHeal ? '💚' : '⚔'} ${ability.name} ${ability.effect.isHeal ? '治疗' : '命中'} ${target?.name ?? user.name}: ${finalValue} ${ability.element ? ELEMENT_LABELS[ability.element] : '物理'}`,
    data: { baseValue, finalValue, special: specialResult },
    timestamp: Date.now(),
  });
  for (const m of specialResult.extra) log.push({ kind: 'action', round: state.round, turn: state.turn, message: m, timestamp: Date.now() });

  return { log, buffTicks: [], ended: false };
}
```

### 5.2 抗性公式 helper

```typescript
// services/abilities/abilityUtils.ts (新增)

import type { Element, ElementalResistances } from '../../types/character';

export function applyResistance(
  baseDamage: number,
  element: Element | null,
  targetResistances: ElementalResistances,
): number {
  if (element === null) return baseDamage;  // 物理, 走原公式
  const r = targetResistances[element];
  return Math.max(0, Math.round(baseDamage * (1 - Math.max(-100, Math.min(100, r)) / 100)));
}
```

### 5.3 战技 special 实现

```typescript
// services/abilities/abilityUtils.ts (新增)

import type { BattleArtSpecial } from '../../types/ability';
import { useCombatStore } from '../../stores/combatStore';
import type { Combatant } from '../combat/types';

/** applySpecial: 战技特殊效果
 * currentTurn 用于 buff.appliedAtTurn 字段 */
export function applySpecial(
  special: BattleArtSpecial | undefined,
  baseDamage: number,
  attacker: Combatant,
  target: Combatant | null,
  currentTurn: number,
): { damage: number; extra: string[] } {
  if (!special) return { damage: baseDamage, extra: [] };
  switch (special) {
    case 'high_crit':
      return { damage: Math.round(baseDamage * 1.3), extra: [`${attacker.name} 命中要害!`] };
    case 'armor_pierce':
      // 实际 defense 减半 在 resolveAbility 主流程 中通过 effective 公式实现
      return { damage: baseDamage, extra: ['穿甲效果生效'] };
    case 'life_steal': {
      const steal = Math.max(1, Math.round(baseDamage * 0.3));
      useCombatStore.getState().applyHeal(attacker.id, steal);
      return { damage: baseDamage, extra: [`${attacker.name} 吸取 ${steal} HP`] };
    }
    case 'self_dodge_penalty':
      useCombatStore.getState().addBuff(attacker.id, {
        ref: 'self_dodge_penalty',
        stacks: 1,
        remainingTurns: 1,
        source: attacker.id,
        appliedAtTurn: currentTurn,
        modifiers: { DEX: -2 },
      });
      return { damage: baseDamage, extra: [`${attacker.name} 体力消耗, 防御下降`] };
  }
}
```

**注**: armor_pierce 的"defense 减半"在 resolveAbility 主流程中实现, 通过 `effectiveAttribute(target, 'DEF')` 读取 target.defense 时乘 0.5。

## §9 错误处理

| 场景 | 处理 |
|---|---|
| `getAbility(id)` 未知 id | 返 null + 战斗 log "[错误] 未知 ability" |
| `ability.cost.ap > user.ap` | 不执行, log "AP 不足" |
| `ability.cost.mp > user.mp` | 不执行, log "MP 不足" |
| 目标已死 | 不执行, log "目标已死" |
| SkillPickerPopover 无任何已学 | 显示"你尚未学习任何{school}能力", tab 仍可切 |
| Combatant 初始化时 characterStore 空 | 抗性默认 ZERO, 不抛错 |
| QTE 关闭时 ability | 走 QTE_NOOP, modifier=0 |
| AOE ability ('all_enemies') | v0.6.2 暂不支持, log "AOE 暂未实现" |
