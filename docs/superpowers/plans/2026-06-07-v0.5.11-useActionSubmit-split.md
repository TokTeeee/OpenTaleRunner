import { useCallback } from 'react';
import { useGameStore } from '../../stores/gameStore';
import { absurdityToLC } from '../../utils/dice';
import {
  _judgeSystem,
  estimateAbsurdity,
  findBestSkill,
} from './shared';
import type { Character, AttributeName } from '../../types/character';
import type { JudgeParams } from '../../types/game';

export interface DiceJudgeResult {
  auto: boolean;
  outcome: 'success' | 'failure' | 'critical';
  reason: string;
  difficultyLC: number;
  autoReason?: string;
}

/**
 * Dice/judgment abstraction. Used by both single & multiplayer submit paths.
 * - absurdity <= 2: auto-success
 * - absurdity > 2: evaluate via _judgeSystem
 */
export function useDiceJudge() {
  const judgeAction = useCallback((action: string, char: Character | null): DiceJudgeResult | null => {
    if (!char) return null;
    const game = useGameStore.getState();
    const absurdity = estimateAbsurdity(action, char);
    if (absurdity <= 2) {
      return {
        auto: true,
        outcome: 'success',
        reason: '无需检定',
        difficultyLC: absurdityToLC(absurdity),
        autoReason: 'low_absurdity',
      };
    }
    const bestSkill = findBestSkill(action, char);
    const judgeParams: JudgeParams = {
      absurdityLevel: absurdity,
      difficultyLC: absurdityToLC(absurdity),
      reason: '本地判定',
      relevantSkill: bestSkill?.name || null,
      relevantAttribute: (bestSkill?.relatedAttribute as AttributeName) || 'WIS',
    };
    return _judgeSystem.evaluate(judgeParams, char, {
      worldDay: game.currentDay,
      region: game.currentRegion,
      subRegion: game.currentSubRegion,
      coordinates: game.coordinates,
      terrain: game.terrain,
      weather: game.weather,
      factions: [],
    });
  }, []);

  return { judgeAction };
}
```

import { useCallback } from 'react';
import { logger } from '../../utils/logger';

export interface ErrorRecoveryDeps {
  setWaitingForPM: (v: boolean) => void;
  setError: (msg: string) => void;
  clearError: () => void;
  applyFallback: () => void;
}

export function useErrorRecovery(deps: ErrorRecoveryDeps) {
  const handlePMError = useCallback((err: unknown, context: string) => {
    const msg = err instanceof Error ? err.message : String(err);
    logger.error(`[useActionSubmit] PM error in ${context}: ${msg}`);
    deps.setWaitingForPM(false);
    deps.setError(`${context} 失败: ${msg}`);
    deps.applyFallback();
  }, [deps]);

  const clearError = useCallback(() => {
    deps.clearError();
  }, [deps]);

  return { handlePMError, clearError };
}
```

import { getPmEngine } from './shared';
import { useDiceJudge } from './useDiceJudge';
import { applyConsequences } from '../../services/consequence/applyConsequences';
import { eventBus } from '../../services/event/EventBus';
import { EVENTS } from '../../services/event/events';

export interface SingleSubmitDeps {
  // ... 待按实际依赖填
}

export function useSingleSubmit(deps: SingleSubmitDeps) {
  const { judgeAction } = useDiceJudge();

  const submitActionSingle = useCallback(async (action: string) => {
    const char = deps.getChar();
    const judgeResult = judgeAction(action, char);
    const engine = getPmEngine();
    const result = await engine.processAction({
      action,
      judgeResult,
      char,
    });
    if (result.consequences) {
      const { newDiscoveries } = applyConsequences(result.consequences) || {};
      if (newDiscoveries?.length) {
        eventBus.emit(EVENTS.DISCOVERY_RECORDED, { items: newDiscoveries });
      }
    }
    return result;
  }, [deps, judgeAction]);

  return { submitActionSingle };
}

```typescript
import { useCallback } from 'react';
import { useSingleSubmit } from './useSingleSubmit';
import { useMultiplayerSubmit } from './useMultiplayerSubmit';
import { useErrorRecovery } from './useErrorRecovery';
// ... 必要 import

export interface ActionSubmitDeps {
  // 全部 deps 集中 (从 usePMInitialization 传)
}

export function useActionSubmit(deps: ActionSubmitDeps) {
  const { submitActionSingle } = useSingleSubmit(deps);
  const { submitActionMulti, skipRound } = useMultiplayerSubmit(deps);
  const { handlePMError, clearError } = useErrorRecovery(deps);

  const submitAction = useCallback(async (action: string) => {
    try {
      if (deps.mode === 'multiplayer') {
        return await submitActionMulti(action);
      }
      return await submitActionSingle(action);
    } catch (err) {
      handlePMError(err, 'submitAction');
    }
  }, [deps.mode, submitActionSingle, submitActionMulti, handlePMError]);

  const submitCustom = useCallback(async (text: string) => {
    return submitAction(text);
  }, [submitAction]);

  const pickChoice = useCallback((choice: Choice) => {
    return submitAction(choice.text);
  }, [submitAction]);

  const abort = useCallback(() => {
    deps.setWaitingForPM(false);
    clearError();
  }, [deps, clearError]);

  const skipAction = skipRound; // multiplayer only

  return { submitAction, submitCustom, pickChoice, abort, skipAction };
}
```
