**主入口顺序**: attributes → conditions → skills → reputation → currency → items

```typescript
/**
 * 从 ItemGainedData 构造 Item,处理升级/新建两种情况.
 * 升级时保留原 itemId (库存去重/装备引用稳定).
 */
export function buildItemFromGained(gained: ItemGainedData, oldItem: Item | null, now: string): Item {
  // 完整实现 (从原 applyConsequences.ts 搬)
  const history: ItemHistoryEntry[] = oldItem ? [
    ...(oldItem.history || []),
    {
      timestamp: now,
      event: 'upgraded',
      description: gained.description || `升级为${gained.name}`,
      oldName: oldItem.name,
      oldDescription: oldItem.description,
    },
  ] : [{
    timestamp: now,
    event: 'acquired',
    description: gained.description || `获得了${gained.name}`,
  }];

  const itemId = oldItem?.itemId
    || (!gained.replacesItemId ? `item_${Date.now()}_${Math.random().toString(36).slice(2, 8)}` : gained.replacesItemId);

  return {
    itemId,
    name: gained.name,
    category: (gained.category as ItemCategory) || 'consumable',
    subCategory: gained.subCategory || '',
    quality: (gained.quality as ItemQuality) || '普通',
    quantity: gained.quantity || 1,
    description: gained.description || '',
    effects: (gained.effects || []).map((e, i) => ({
      id: `eff_${Date.now()}_${i}`,
      type: (e.type as ItemEffect['type']) || 'special',
      value: e.value || 0,
      description: e.description || '',
    })),
    value: 0,
    durability: oldItem?.durability,
    maxDurability: oldItem?.maxDurability,
    history,
    createdAt: oldItem?.createdAt || now,
    source: oldItem?.source || '',
    equipped: oldItem?.equipped,
    equipSlot: oldItem?.equipSlot,
    canBeEquipped: gained.category === 'weapon' || gained.category === 'armor' || gained.category === 'accessory',
    canBeUsed: gained.category === 'consumable',
  };
}

/**
 * 从 itemRegistry 重新拼装 character 的背包视图.
 * 旧版 UI 消费 char.inventory.backpack,这里作为派生层保持兼容.
 */
export function syncBackpackFromRegistry(characterId: string): void {
  const charStore = useCharacterStore.getState();
  const char = charStore.character;
  if (!char) return;
  const registry = useItemRegistryStore.getState();
  const items = registry.byHolder({ kind: 'character', refId: characterId })
    .filter((it) => it.holder !== null)
    .map((it) => worldItemToLegacyView(it));
  charStore.updateInventory({ ...char.inventory, backpack: items });
}

    subCategory: w.subCategory || '',

    maxDurability: w.durability?.max ?? w.maxDurability,
    history: w.history,
    createdAt: w.createdAt,
    source: w.source,
    equipped: w.equipped,
    equipSlot: w.equipSlot,
    canBeEquipped: w.canBeEquipped,
    canBeUsed: w.canBeUsed,
  };
}
```

```typescript
import { useCharacterStore } from '../../stores/characterStore';
import type { ConsequenceData } from '../../types/game';

/**
 * 业务域 1: attributes
 * 改"我是谁"的基础数值 + 身份
 *
 * 二级字段函数 (模块内私有):
 *   - applyAttributeChanges(cons.attributeChanges)
 *   - applyIdentityChanges(cons.identityChanges)
 */
export function applyAttributes(cons: Pick<ConsequenceData, 'attributeChanges' | 'identityChanges'>): void {
  try {
    if (cons.attributeChanges) applyAttributeChanges(cons.attributeChanges);
    if (cons.identityChanges) applyIdentityChanges(cons.identityChanges);
  } catch (err) {
    console.warn(`[applyAttributes] failed: ${err instanceof Error ? err.message : String(err)}`);
  }
}

function applyAttributeChanges(changes: Record<string, number>): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const newAttrs = { ...char.attributes };
  for (const [k, v] of Object.entries(changes)) {
    newAttrs[k as keyof typeof newAttrs] = (newAttrs[k as keyof typeof newAttrs] || 0) + v;
  }
  useCharacterStore.getState().updateBaseAttributes(newAttrs);
}

function applyIdentityChanges(identity: { name?: string; race?: string; background?: string }): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const newIdentity = { ...char.identity, ...identity };
  useCharacterStore.getState().updateIdentity(newIdentity);
}
```

```typescript
import { useCharacterStore } from '../../stores/characterStore';
import type { ConsequenceData } from '../../types/game';

/**
 * 业务域 2: conditions
 * 改"我身上挂的状态"
 */
export function applyConditions(cons: Pick<ConsequenceData, 'conditionsAdded' | 'conditionsRemoved'>): void {
  try {
    if (cons.conditionsAdded?.length) applyConditionsAdded(cons.conditionsAdded);
    if (cons.conditionsRemoved?.length) applyConditionsRemoved(cons.conditionsRemoved);
  } catch (err) {
    console.warn(`[applyConditions] failed: ${err instanceof Error ? err.message : String(err)}`);
  }
}

function applyConditionsAdded(added: { id: string; name: string }[]): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const existing = new Set(char.conditions?.map(c => c.id) || []);
  const newOnes = added.filter(c => !existing.has(c.id));
  useCharacterStore.getState().updateConditions([...(char.conditions || []), ...newOnes]);
}

function applyConditionsRemoved(removedIds: string[]): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const removeSet = new Set(removedIds);
  const newOnes = (char.conditions || []).filter(c => !removeSet.has(c.id));
  useCharacterStore.getState().updateConditions(newOnes);
}
```

```typescript
import { useCharacterStore } from '../../stores/characterStore';
import type { ConsequenceData } from '../../types/game';

/**
 * 业务域 3: skills
 * 改"我会什么"
 */
export function applySkills(cons: Pick<ConsequenceData, 'skillsModified'>): void {
  try {
    if (cons.skillsModified?.length) applySkillsModified(cons.skillsModified);
  } catch (err) {
    console.warn(`[applySkills] failed: ${err instanceof Error ? err.message : String(err)}`);
  }
}

function applySkillsModified(modified: any[]): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const skillMap = new Map((char.skills || []).map(s => [s.id, s]));
  for (const m of modified) {
    const existing = skillMap.get(m.id);
    if (existing) {
      skillMap.set(m.id, { ...existing, ...m });
    } else {
      skillMap.set(m.id, m);
    }
  }
  useCharacterStore.getState().updateSkills(Array.from(skillMap.values()));
}
```

```typescript
    if (cons.currencyChange) applyCurrencyChange(cons.currencyChange);
    if (cons.attributeChanges?.CHA && cons.attributeChanges.CHA > 0) {
      applyCharismaRedirect(cons.attributeChanges.CHA);
    }
  } catch (err) {
    console.warn(`[applyReputation] failed: ${err instanceof Error ? err.message : String(err)}`);
  }
}

function applyReputationChange(changes: Record<string, number>): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const newRep = { ...char.reputation };
  for (const [k, v] of Object.entries(changes)) {
    newRep[k] = (newRep[k] || 0) + v;
  }
  useCharacterStore.getState().updateReputation(newRep);
}

function applyCurrencyChange(amount: number): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  useCharacterStore.getState().updateCurrency((char.currency || 0) + amount);
}

/**
 * CHA 提升 -> 所有 faction reputation +CHA 增量.
 * 反映"个人魅力提升会提升所有人对你的看法".
 */
function applyCharismaRedirect(chaDelta: number): void {
  const char = useCharacterStore.getState().character;
  if (!char) return;
  const newRep: Record<string, number> = { ...char.reputation };
  for (const k of Object.keys(newRep)) {
    newRep[k] = (newRep[k] || 0) + chaDelta;
  }
  useCharacterStore.getState().updateReputation(newRep);
}
```

```typescript
import { useItemRegistryStore } from '../../stores/itemRegistryStore';
import { useCodexStore } from '../../stores/codexStore';
import { useCharacterStore } from '../../stores/characterStore';
import { buildItemFromGained, syncBackpackFromRegistry, findInRegistryByCharacter } from './helpers';
import type { ConsequenceData } from '../../types/game';
import type { Item } from '../../types/item';

export interface ItemDiscovery {
  itemId: string;
  itemName: string;
  discoveredAt: string;
}

/**
 * 业务域 5: items
 * 改"我拥有什么"
 *
 * 本域是唯一会产生 newDiscoveries 的域
 */
export function applyItems(cons: Pick<ConsequenceData, 'itemsGained' | 'itemsLost' | 'itemsModified'>): ItemDiscovery[] {
  const discoveries: ItemDiscovery[] = [];
  try {
    if (cons.itemsGained?.length) discoveries.push(...applyItemsGained(cons.itemsGained));
    if (cons.itemsLost?.length) applyItemsLost(cons.itemsLost);
    if (cons.itemsModified?.length) applyItemsModified(cons.itemsModified);
  } catch (err) {
    console.warn(`[applyItems] failed: ${err instanceof Error ? err.message : String(err)}`);
  }
  return discoveries;
}

function applyItemsGained(gained: any[]): ItemDiscovery[] {
  const discoveries: ItemDiscovery[] = [];
  const charId = useCharacterStore.getState().character?.id;
  if (!charId) return discoveries;
  const now = new Date().toISOString();
  const registry = useItemRegistryStore.getState();

  for (const g of gained) {
    const oldItem = findInRegistryByCharacter(charId).find(it => it.itemId === g.replacesItemId) || null;
    const item = buildItemFromGained(g, oldItem as Item | null, now);
    registry.register({
      ...item,
      holder: { kind: 'character', refId: charId },
    });
    discoveries.push({ itemId: item.itemId, itemName: item.name, discoveredAt: now });
    // record discovery
    useCodexStore.getState().recordDiscovery?.({ itemId: item.itemId, itemName: item.name, timestamp: now });
  }

  syncBackpackFromRegistry(charId);
  return discoveries;
}

function applyItemsLost(itemIds: string[]): void {
  const registry = useItemRegistryStore.getState();
  for (const id of itemIds) {
    registry.destroy?.(id);
  }
  const charId = useCharacterStore.getState().character?.id;
  if (charId) syncBackpackFromRegistry(charId);
}

function applyItemsModified(modified: any[]): void {
  const registry = useItemRegistryStore.getState();
  for (const m of modified) {
    registry.patch?.(m.itemId, m);
  }
  const charId = useCharacterStore.getState().character?.id;
  if (charId) syncBackpackFromRegistry(charId);
}
```

```typescript
import type { ConsequenceData } from '../../types/game';
import { applyAttributes } from './applyAttributes';
import { applyConditions } from './applyConditions';
import { applySkills } from './applySkills';
import { applyReputation } from './applyReputation';
import { applyItems, type ItemDiscovery } from './applyItems';
import { logger } from '../../utils/logger';

export interface ApplyConsequencesResult {
  newDiscoveries: ItemDiscovery[];
}

/**
 * 主入口: 按业务域顺序应用 consequence.
 * 业务域顺序 (约束):
 *   1. attributes - 基础值先变
 *   2. conditions - 状态效果在属性之上
 *   3. skills     - 能力依赖属性
 *   4. reputation - 含 CHA 重定向 (依赖 attributes)
 *   5. items      - 依赖其他域的 final state, 最后
 *
 * 错误隔离: 每个业务域独立 try/catch, 错误不阻断后续域.
 */
export function applyConsequences(
  cons: ConsequenceData,
  opts?: { silent?: boolean },
): ApplyConsequencesResult | undefined {
  if (!cons) return undefined;
  try {
    applyAttributes(cons);
    applyConditions(cons);
    applySkills(cons);
    applyReputation(cons);
    const newDiscoveries = applyItems(cons);
    return { newDiscoveries };
  } catch (err) {
    if (!opts?.silent) {
      logger.error(`[applyConsequences] failed: ${err instanceof Error ? err.message : String(err)}`);
    }
    return undefined;
  }
}
```
