# v0.6.2 Combat & Ability Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 v0.6.2 — 统一 Ability 数据模型(覆盖 v0.6.1 双类型) + 16 个初始能力(6 魔法 + 6 祷告 + 4 战技) + 战斗接入(ActionMenu 6 按钮 + SkillPickerPopover + ActionResolver.resolveAbility + 抗性公式 + 战技 special)。

**Tech Stack:** TypeScript (strict mode) + Zustand stores + React + Vitest + ESLint。沿用现有 `types/character.ts` / `stores/characterStore.ts` / `services/combat/ActionResolver.ts` 等。

```typescript
// client/src/types/ability.ts (整文件替换)
import type { AttributeName, Attributes } from './character';

export type MagicElement = 'fire' | 'ice' | 'lightning' | 'wind' | 'earth' | 'arcane';
export type PrayerElement = 'holy' | 'shadow';
export type Element = MagicElement | PrayerElement;
export type AbilitySchool = 'magic' | 'prayer' | 'battle_art';
export type AbilityTarget =
  | 'enemy' | 'ally' | 'self' | 'all_enemies' | 'all_allies';
export type AbilityTier = 1 | 2 | 3;

export const ELEMENT_LABELS: Record<Element, string> = {
  fire: '火', ice: '冰', lightning: '雷', wind: '风',
  earth: '土', arcane: '奥术', holy: '神圣', shadow: '暗影',
};

export const ELEMENT_ICONS: Record<Element, string> = {
  fire: '🔥', ice: '❄️', lightning: '⚡', wind: '🌪️',
  earth: '⛰️', arcane: '🔮', holy: '✨', shadow: '🌑',
};

export const SCHOOL_LABELS: Record<AbilitySchool, string> = {
  magic: '魔法', prayer: '祷告', battle_art: '战技',
};

export const SCHOOL_ICONS: Record<AbilitySchool, string> = {
  magic: '✨', prayer: '🌟', battle_art: '⚔️',
};

export interface AbilityDescription {
  shortEffect: string;
  narrative: string;
  visualTag: string;
}

export type BattleArtSpecial =
  | 'high_crit'
  | 'armor_pierce'
  | 'life_steal'
  | 'self_dodge_penalty';

export interface AbilityEffect {
  damageDice: string;
  isHeal: boolean;
  attributeScale: AttributeName;
  element: Element | null;
  applyBuff?: { ref: string; stacks: number; turns: number };
  special?: BattleArtSpecial;
}

export interface AbilityRequirement {
  classes: Array<'warrior' | 'mage' | 'cleric' | 'thief' | 'any'>;
  minAttribute: Partial<Attributes>;
  minLevel: number;
}

export interface Ability {
  id: string;
  name: string;
  school: AbilitySchool;
  element: Element | null;
  tier: AbilityTier;
  requirements: AbilityRequirement;
  cost: { ap: number; mp: number };
  target: AbilityTarget;
  effect: AbilityEffect;
  description: AbilityDescription;
}

export type { AttributeName } from './character';
```

```typescript
// client/src/services/abilities/learnAbility.ts (整文件替换)
import type { Character, AttributeName, Attributes } from '../../types/character';
import type { Ability } from '../../types/ability';

export interface LearnCheckInput {
  character: Pick<Character, 'classId' | 'level' | 'attributes'>;
  ability: Ability;
}

export type LearnCheckResult =
  | { canLearn: true }
  | { canLearn: false; reason: 'class' | 'attribute' | 'level'; required: string };

export function checkCanLearn(input: LearnCheckInput): LearnCheckResult {
  const { character, ability } = input;
  const r = ability.requirements;

  // 1. 职业
  const allowedClasses = r.classes;
  const classOk =
    allowedClasses.includes('any') ||
    (character.classId !== null && allowedClasses.includes(character.classId as 'mage' | 'cleric' | 'warrior' | 'thief'));
  if (!classOk) {
    return { canLearn: false, reason: 'class', required: allowedClasses.join(' 或 ') };
  }

  // 2. 属性 (Partial<Attributes> 任一项 ≥ 门槛)
  for (const [attr, threshold] of Object.entries(r.minAttribute) as [AttributeName, number][]) {
    const v = character.attributes[attr];
    if (v < threshold) {
      return { canLearn: false, reason: 'attribute', required: `${attr} ≥ ${threshold}` };
    }
  }

  // 3. 等级
  if (character.level < r.minLevel) {
    return { canLearn: false, reason: 'level', required: `等级 ≥ ${r.minLevel}` };
  }

  return { canLearn: true };
}
```

```typescript
import type { Element } from '../../types/ability';
import type { ElementalResistances } from '../../types/character';
import { rollDice } from '../../utils/dice';

type RollFn = (sides: number, count: number) => number[];

/** 解析骰公式 '1d6' / '1d6+2' / '-1d6-1' / '0', 返 { total, rolls } */
export function parseDiceFormula(formula: string, roll: RollFn): { total: number; rolls: number[] } {
  if (formula === '0') return { total: 0, rolls: [] };
  // 提取符号 + 骰子 + 加成
  const match = formula.match(/^(-?)(\d+)d(\d+)([+-]\d+)?$/);
  if (!match) throw new Error(`Invalid dice formula: ${formula}`);
  const sign = match[1] === '-' ? -1 : 1;
  const count = parseInt(match[2], 10);
  const sides = parseInt(match[3], 10);
  const bonus = match[4] ? parseInt(match[4], 10) : 0;
  const rolls = roll(sides, count);
  const sum = rolls.reduce((a, b) => a + b, 0);
  return { total: Math.max(0, sign * (sum + bonus)), rolls };
}

/** 抗性 → 伤害乘数. 公式: 1 - resistance/100, clamp [-100, 100] */
export function resistanceMultiplier(resistance: number): number {
  const clamped = Math.max(-100, Math.min(100, resistance));
  return 1 - clamped / 100;
}

/** 应用抗性. element=null (物理) 不应用. damage 不可 < 0 */
export function applyResistance(
  baseDamage: number,
  element: Element | null,
  targetResistances: ElementalResistances,
): number {
  if (element === null) return baseDamage;
  const r = targetResistances[element];
  return Math.max(0, Math.round(baseDamage * resistanceMultiplier(r)));
}
```

```typescript
import type { BattleArtSpecial } from '../../types/ability';
import type { Combatant } from '../combat/types';
import { useCombatStore } from '../../stores/combatStore';

export function applySpecial(
  special: BattleArtSpecial | undefined,
  baseDamage: number,
  attacker: Combatant,
  target: Combatant | null,
  currentTurn: number,
): { damage: number; extra: string[] } {
  if (!special) return { damage: baseDamage, extra: [] };

    case 'high_crit':

      useCombatStore.getState().applyHeal(attacker.id, steal);

        ref: 'self_dodge_penalty',
        stacks: 1,
        remainingTurns: 1,
        source: attacker.id,
        appliedAtTurn: currentTurn,
        modifiers: { DEX: -2 },
      });
      return { damage: baseDamage, extra: [`${attacker.name} 体力消耗, 防御下降`] };
  }
```

```typescript
private resolveAbility(
  action: Extract<CombatAction, { kind: 'ability' }>,
  ctx: ResolverContext,
  state: CombatState,
): TurnResult {
  const user = state.combatants[action.userId];
  const target = action.targetId ? state.combatants[action.targetId] : null;
  const ability = getAbility(action.abilityId);
  const log: CombatLogEntry[] = [];

  if (!ability) {
    return { log: [this.errorLog('未知 ability', state)], buffTicks: [], ended: false };
  }
  if (!user || user.isDead) {
    return { log: [this.errorLog('无效 user', state)], buffTicks: [], ended: false };
  }
  if (ability.cost.ap > user.ap) {
    return { log: [this.errorLog('AP 不足', state)], buffTicks: [], ended: false };
  }
  if (ability.cost.mp > (user.mp ?? 0)) {
    return { log: [this.errorLog('MP 不足', state)], buffTicks: [], ended: false };
  }
  if (target && target.isDead) {
    return { log: [this.errorLog('目标已死', state)], buffTicks: [], ended: false };
  }
  if (ability.target === 'all_enemies' || ability.target === 'all_allies') {
    return { log: [this.errorLog('AOE 暂未实现', state)], buffTicks: [], ended: false };
  }

  // 扣 AP + MP
  useCombatStore.getState().applyAP(user.id, -ability.cost.ap);
  if (ability.cost.mp > 0) useCombatStore.getState().applyMP(user.id, -ability.cost.mp);

  // 命中投
  const toHit = rollToHit(user, ctx.roll);
  const threshold = target ? hitThreshold(target, 0, false) : { total: 0 };
  const hit = !target || checkHit(toHit.total, threshold.total);

  log.push({
    kind: 'action',
    round: state.round,
    turn: state.turn,
    message: `${user.name} 释放 ${SCHOOL_LABELS[ability.school]} · ${ability.name} → ${target?.name ?? '自身'}: d20=${toHit.d20}+${toHit.dexMod}=${toHit.total} vs ${threshold.total}`,
    data: { ability, toHit, threshold },
    timestamp: Date.now(),
  });

  if (!hit && target) {
    log.push({ kind: 'action', round: state.round, turn: state.turn, message: `${ability.name} 未命中 ${target.name}`, timestamp: Date.now() });
    return { log, buffTicks: [], ended: false };
  }

  // 算 baseValue
  const diceResult = parseDiceFormula(ability.effect.damageDice, ctx.roll);
  const attrValue = effectiveAttribute(user, ability.effect.attributeScale);
  const attrMod = Math.floor((attrValue - 10) / 2);
  const baseValue = Math.max(0, diceResult.total + attrMod);

  // 战技 special
  const specialResult = applySpecial(ability.effect.special, baseValue, user, target, state.turn);
  let finalValue = specialResult.damage;

  // armor_pierce 实际应用 (target.defense 减半, 简化: 直接加 25% 伤害)
  // 注: 若 v0.6.2 后期改, 此处需调整. 当前用 +25% 简化.
  if (ability.effect.special === 'armor_pierce') {
    finalValue = Math.round(finalValue * 1.25);
  }

  // 抗性 (magic/prayer)
  if (ability.element !== null && target) {
    finalValue = applyResistance(finalValue, ability.element, target.elementalResistances);
  }

  // QTE 缩放
  const qteRes = ctx.qte({ action, attacker: user, target: target!, state });
  finalValue = Math.max(0, Math.round(finalValue * (1 + qteRes.modifier * 0.5)));

  // isHeal 决定加血/扣血
  if (ability.effect.isHeal) {
    useCombatStore.getState().applyHeal(target?.id ?? user.id, finalValue);
  } else if (target) {
    useCombatStore.getState().applyDamage(target.id, finalValue);
  }

  // buff
  if (ability.effect.applyBuff && target) {
    useCombatStore.getState().addBuff(target.id, {
      ref: ability.effect.applyBuff.ref,
      stacks: ability.effect.applyBuff.stacks,
      remainingTurns: ability.effect.applyBuff.turns,
      source: user.id,
      appliedAtTurn: state.turn,
      modifiers: {},
    });
  }

  log.push({
    kind: 'action',
    round: state.round,
    turn: state.turn,
    message: `${ability.effect.isHeal ? '💚' : '⚔'} ${ability.name} ${ability.effect.isHeal ? '治疗' : '命中'} ${target?.name ?? user.name}: ${finalValue} ${ability.element ? ELEMENT_LABELS[ability.element] : '物理'}`,
    data: { baseValue, finalValue, special: specialResult },
    timestamp: Date.now(),
  });
  for (const m of specialResult.extra) {
    log.push({ kind: 'action', round: state.round, turn: state.turn, message: m, timestamp: Date.now() });
  }

  return { log, buffTicks: [], ended: false };
}

private errorLog(message: string, state: CombatState): CombatLogEntry {
  return {
    kind: 'action',
    round: state.round,
    turn: state.turn,
    message: `[错误] ${message}`,
    timestamp: Date.now(),
  };
```

```tsx
// client/src/components/combat/SkillPickerPopover.tsx
import { createPortal } from 'react-dom';
import { useState, useMemo } from 'react';
import {
  ABILITY_REGISTRY, getAbility,
} from '../../data/abilities';
import { SCHOOL_LABELS, SCHOOL_ICONS, ELEMENT_ICONS } from '../../types/ability';
import type { AbilitySchool } from '../../types/ability';

interface SkillPickerPopoverProps {
  isOpen: boolean;
  onClose: () => void;
  learnedAbilityIds: string[];
  onSelect: (abilityId: string) => void;
}

const SCHOOLS: AbilitySchool[] = ['magic', 'prayer', 'battle_art'];

export function SkillPickerPopover({ isOpen, onClose, learnedAbilityIds, onSelect }: SkillPickerPopoverProps) {
  const learnedSet = useMemo(() => new Set(learnedAbilityIds), [learnedAbilityIds]);
  const firstLearnedSchool = useMemo(() => {
    for (const id of learnedAbilityIds) {
      const a = getAbility(id);
      if (a) return a.school;
    }
    return 'magic' as AbilitySchool;
  }, [learnedAbilityIds]);

  const [tab, setTab] = useState<AbilitySchool>(firstLearnedSchool);

  if (!isOpen) return null;

  const allInTab = Object.values(ABILITY_REGISTRY).filter(a => a.school === tab);
  const learnedInTab = allInTab.filter(a => learnedSet.has(a.id));
  const unlearnedInTab = allInTab.filter(a => !learnedSet.has(a.id));

  return createPortal(
    <div className="skill-picker-overlay" onClick={onClose} data-testid="skill-picker">
      <div className="skill-picker-modal" onClick={e => e.stopPropagation()}>
        <div className="skill-picker-header">
          {SCHOOLS.map(s => (
            <button
              key={s}
              className={`skill-picker-tab ${tab === s ? 'active' : ''}`}
              onClick={() => setTab(s)}
              data-testid={`tab-${s}`}
            >
              {SCHOOL_ICONS[s]} {SCHOOL_LABELS[s]}
            </button>
          ))}
          <button className="skill-picker-close" onClick={onClose} data-testid="close">✕</button>
        </div>

        <div className="skill-picker-body">
          {learnedInTab.length === 0 && (
            <p className="skill-picker-empty" data-testid="empty">
              你尚未学习任何{SCHOOL_LABELS[tab]}能力, 满足条件后可学习。
            </p>
          )}

          {learnedInTab.length > 0 && (
            <>
              <h4>已学 ({learnedInTab.length}):</h4>
              {learnedInTab.map(a => (
                <div key={a.id} className="skill-picker-card" data-testid={`learned-${a.id}`}>
                  <span className="card-icon">
                    {a.element ? ELEMENT_ICONS[a.element] : SCHOOL_ICONS[a.school]}
                  </span>
                  <span className="card-name">{a.name}</span>
                  <span className="card-effect">{a.description.shortEffect}</span>
                  <span className="card-cost">
                    {a.cost.ap}AP {a.cost.mp > 0 ? `${a.cost.mp}MP` : ''}
                  </span>
                  <button onClick={() => onSelect(a.id)} data-testid={`select-${a.id}`}>选择</button>
                </div>
              ))}
            </>
          )}

          {unlearnedInTab.length > 0 && (
            <>
              <h4>未学 ({unlearnedInTab.length}):</h4>
              {unlearnedInTab.map(a => (
                <div key={a.id} className="skill-picker-card disabled" data-testid={`unlearned-${a.id}`}>
                  <span className="card-icon">
                    {a.element ? ELEMENT_ICONS[a.element] : SCHOOL_ICONS[a.school]}
                  </span>
                  <span className="card-name">{a.name}</span>
                  <span className="card-effect">
                    {a.description.shortEffect} (需 Lv.{a.requirements.minLevel} +
                    {Object.entries(a.requirements.minAttribute).map(([k, v]) => `${k} ${v}`).join(', ')})
                  </span>
                </div>
              ))}
            </>
          )}
        </div>
      </div>
    </div>,
    document.body,
  );
}
```
