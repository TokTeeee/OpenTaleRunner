# v0.5.14 — Character Creation + CharacterPanel 体验修订 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 修订 4 处用户体验问题：Wizard 7 步重排、CharacterPanel 属性栏高度优化、职业显示在姓名下并可点击、技能合并。

**Architecture:**
- **Wizard**: 内部 `renderStep` switch 重新映射 case 编号；新增 `renderNameAppearance()` 从 `renderReview()` 抽出；删除 `renderReview()`；背景 prompt 升级带 class+attrs。
- **CharacterPanel**: 头部加职业文字（点击触发 Modal）；属性 section 高度优化（雷达 140×100 + 6 行 chip）；删除独立 `ClassSkillTreeView`；新建 `SkillsSection` 合并 3 种 chip；声望/装备/货币/背包改 `<details>` 折叠。
- **新组件**: `ClassSkillTreeModal`（全屏 12 节点大网格）；`SkillsSection`（3 色 chip）。
- **state 不改**: `useCharacterStore` 不动；CharacterPanel 顶层 1 个 `useState<boolean>` for modal。

**Tech Stack:** TypeScript + React + zustand + vitest + @testing-library/react + TailwindCSS

新建 `client/src/components/panels/CharacterPanel/SkillsSection.tsx`：

```typescript
import { useCharacterStore } from '../../../stores/characterStore';
import { getClass } from '../../../data/classes';

type ChipType = 'origin' | 'classlearned' | 'classavailable';

type SkillChip = {
  key: string;
  type: ChipType;
  label: string;
};

const TYPE_STYLES: Record<ChipType, string> = {
  origin: 'bg-indigo-500/10 border border-indigo-500/20 text-indigo-300',
  classlearned: 'bg-emerald-500/10 border border-emerald-500/20 text-emerald-300',
  classavailable: 'bg-amber-500/10 border border-amber-500/20 text-amber-300',
};

export function SkillsSection() {
  const character = useCharacterStore((s) => s.character);
  if (!character) return null;

  const chips: SkillChip[] = [];

  // 1. 出身技能 (origin) — 蓝
  for (const s of character.skills ?? []) {
    chips.push({
      key: s.id,
      type: 'origin',
      label: `${s.name} Lv.${s.level}`,
    });
  }

  // 2. 职业相关
  const classDef = character.classId ? getClass(character.classId) : null;
  if (classDef) {
    // 2a. 已学 (classlearned) — 绿
    for (const nodeId of character.classSkills ?? []) {
      const node = (classDef as any).nodes?.find((n: any) => n.nodeId === nodeId);
      if (node) {
        chips.push({
          key: nodeId,
          type: 'classlearned',
          label: `T${node.tier}·${node.slot} ${node.name}`,
        });
      }
    }

    // 2b. 可学未选 (classavailable) — 黄 (按 tier 升序)
    const learnedSet = new Set(character.classSkills ?? []);
    const available = (classDef as any).nodes
      ?.filter((n: any) =>
        !learnedSet.has(n.nodeId) && n.unlockedByLevel <= (character.level ?? 1)
      )
      .sort((a: any, b: any) => a.tier - b.tier || a.slot - b.slot) ?? [];
    for (const node of available) {
      chips.push({
        key: node.nodeId,
        type: 'classavailable',
        label: `T${node.tier}·${node.slot} ${node.name}`,
      });
    }
  }

  if (chips.length === 0) return null;

  return (
    <div>
      <div className="text-[10px] text-gray-600 uppercase tracking-wider mb-1">技能</div>
      <div className="flex flex-wrap gap-1">
        {chips.map((c) => (
          <span
            key={c.key}
            data-testid={`skill-chip-${c.type}-${c.key}`}
            className={`text-[10px] px-2 py-0.5 rounded-md ${TYPE_STYLES[c.type]}`}
            title={c.label}
          >
            {c.label}
          </span>
        ))}
      </div>
    </div>
  );
}
```

```typescript
import { useEffect, useState } from 'react';
import { getClass } from '../../../data/classes';
import type { ClassDefinition, ClassNode } from '../../../types/class';

type Props = {
  classId: string;
  isOpen: boolean;
  onClose: () => void;
  learnedNodes: string[];
  currentLevel: number;
};

export function ClassSkillTreeModal({ classId, isOpen, onClose, learnedNodes, currentLevel }: Props) {
  const [selectedNode, setSelectedNode] = useState<ClassNode | null>(null);
  const classDef = getClass(classId);

  useEffect(() => {
    if (!isOpen) return;
    const handler = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    document.addEventListener('keydown', handler);
    return () => document.removeEventListener('keydown', handler);
  }, [isOpen, onClose]);

  if (!isOpen || !classDef) return null;

  const nodes = (classDef as any).nodes ?? [];
  const learnedSet = new Set(learnedNodes);
  const tierMax = Math.max(...nodes.map((n: ClassNode) => n.tier), 1);

  return (
    <div
      data-testid="class-skill-tree-modal"
      className="fixed inset-0 z-50 bg-black/80 flex items-center justify-center"
      onClick={onClose}
    >
      <div
        className="bg-gray-900 border border-gray-700 rounded-xl w-[90vw] h-[90vh] flex flex-col"
        onClick={(e) => e.stopPropagation()}
      >
        {/* Header */}
        <div className="flex items-center justify-between p-4 border-b border-gray-700">
          <h2 className="text-lg font-bold text-gray-200">
            ⚔️ {classDef.name} 技能树
            <span className="text-sm text-gray-500 ml-3">
              [{learnedNodes.length}/{nodes.length} 已学]
            </span>
          </h2>
          <button
            data-testid="skilltree-close"
            onClick={onClose}
            className="text-gray-500 hover:text-gray-300 text-2xl"
          >
            ✕
          </button>
        </div>

        {/* 12 节点网格 (3 列 × 4 行, 按 tier 排) */}
        <div className="flex-1 overflow-y-auto p-6">
          <div
            className="grid gap-3"
            style={{ gridTemplateColumns: `repeat(3, minmax(0, 1fr))` }}
          >
            {Array.from({ length: tierMax }, (_, i) => i + 1).map((tier) =>
              nodes
                .filter((n: ClassNode) => n.tier === tier)
                .sort((a: ClassNode, b: ClassNode) => a.slot - b.slot)
                .map((node: ClassNode) => {
                  const learned = learnedSet.has(node.nodeId);
                  const unlocked = node.unlockedByLevel <= currentLevel;
                  const bg = learned
                    ? 'bg-emerald-500/20 border-emerald-400/50'
                    : unlocked
                    ? 'bg-white/5 border-white/20'
                    : 'bg-gray-800/50 border-gray-700/50 opacity-60';
                  return (
                    <button
                      key={node.nodeId}
                      data-testid={`skilltree-node-${node.nodeId}`}
                      onClick={() => setSelectedNode(node)}
                      className={`p-4 rounded-lg border-2 text-left transition-all ${bg}`}
                    >
                      <div className="text-[10px] text-gray-500 uppercase">
                        T{node.tier}·{node.slot}
                      </div>
                      <div className="text-sm font-semibold text-gray-200 mt-1">
                        {node.name}
                      </div>
                      {!unlocked && (
                        <div className="text-[10px] text-rose-400 mt-2">
                          🔒 需 Lv.{node.unlockedByLevel}
                        </div>
                      )}
                      {learned && (
                        <div className="text-[10px] text-emerald-400 mt-2">
                          ✅ 已学
                        </div>
                      )}
                    </button>
                  );
                }),
            )}
          </div>
        </div>

        {/* 节点详情面板 */}
        {selectedNode && (
          <div
            data-testid="skilltree-detail"
            className="p-4 border-t border-gray-700 bg-gray-800/50"
          >
            <div className="text-sm font-semibold text-gray-200">
              {selectedNode.name} (T{selectedNode.tier}·{selectedNode.slot})
            </div>
            <div className="text-xs text-gray-400 mt-2">
              {selectedNode.description}
            </div>
            <div className="text-[10px] text-gray-500 mt-1">
              等级要求: Lv.{selectedNode.unlockedByLevel}
              {learnedSet.has(selectedNode.nodeId) ? ' (已达成 ✅)' : ''}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
```
