```typescript
  /**
   * Validate a PromptOverride for the given slot.
   * Returns null on valid, or an error string on invalid.
   *
   * - queryProtocol slot: enforces whitelist of legal fields
   *   (求差集: content 中只能含白名单字段前缀,其他视为非法)
   * - all slots: content length <= 2000 chars
   * - whitelist checked BEFORE length (whitelist error has priority)
   */
  validateOverride(o: PromptOverride, slot: string): string | null {
    if (slot === 'queryProtocol') {
      const WHITELIST = ['SCENE:', 'NPC:', 'QUEST:', 'INVENTORY:', 'COMBAT:', 'TIME:'];
      const hasWhitelisted = WHITELIST.some(prefix => o.content.startsWith(prefix) || o.content.includes('\n' + prefix));
      if (!hasWhitelisted) {
        return 'queryProtocol content must start with one of: ' + WHITELIST.join(', ');
      }
    }
    if (o.content.length > 2000) {
      return `content length ${o.content.length} exceeds max 2000 chars`;
    }
    return null;
  }
```

注意: `PromptOverride` 类型需 import (如未 import,在文件顶部加 `import type { PromptOverride } from '../../types/prompt';`)。

from fastapi import Depends, Header
from ..auth.jwt_utils import decode_token  # 实际路径按项目

@router.post("/rooms")
async def create_room_endpoint(
    req: CreateRoomRequest,
    authorization: str | None = Header(default=None),
):
    player_name = req.player_id  # fallback
    if authorization and authorization.startswith("Bearer "):
        token = authorization[7:]
        try:
            payload = decode_token(token)
            player_name = payload.get("username", req.player_id)
        except Exception:
            pass  # invalid token, fallback
    room = await service.create_room(
        host_id=req.player_id,
        config=req.config,
        mode=req.mode,
        inherit_data=req.inherit_data,
        player_name=player_name,
    )
    return ...
