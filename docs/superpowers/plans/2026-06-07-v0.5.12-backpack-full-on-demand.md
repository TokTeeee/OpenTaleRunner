```typescript
export interface InventorySearchParams {
  keyword: string;
  characterId: string;
}

export interface InventorySearchResult {
  itemId: string;
  name: string;
  category: string;
  quality: string;
  quantity: number;
}

export function inventorySearch(params: InventorySearchParams): InventorySearchResult[] {
  const registry = useItemRegistryStore.getState();
  const results: InventorySearchResult[] = [];
  for (const [itemId, item] of registry.registry.entries()) {
    if (item.holder?.kind !== 'character') continue;
    if (item.holder.refId !== params.characterId) continue;
    if (params.keyword && !item.name.toLowerCase().includes(params.keyword.toLowerCase())) continue;
    results.push({
      itemId: item.itemId,
      name: item.name,
      category: item.category,
      quality: item.quality,
      quantity: item.quantity,
    });
  }
  return results;
}

import { inventorySearch } from '../engine/QueryResolver';

export const inventorySearchTool = {
  name: 'inventory_search',
  description: '搜索角色背包中的物品 (返回匹配的 item 列表)',
  parameters: {
    type: 'object',
    properties: {
      keyword: { type: 'string', description: '搜索关键词 (物品名子串,大小写不敏感,空字符串返回全部)' },
      characterId: { type: 'string', description: '目标角色 ID (省略则用当前玩家)' },
    },
    required: ['keyword'],
  },
  handler: async (params: { keyword: string; characterId?: string }) => {
    const charId = params.characterId || getCurrentCharacterId();
    return inventorySearch({ keyword: params.keyword, characterId: charId });
  },
};

function getCurrentCharacterId(): string {
  // 按项目实际从 characterStore 取
  return useCharacterStore.getState().character?.id || '';
}
```
