```python
# server/repositories/character_repo.py (add at top, after imports)
def _migrate_v04_to_v05(data: dict) -> dict:
    """Idempotent backfill for v0.5 fields. Runs on every create/update."""
    data.setdefault("level", 1)
    data.setdefault("exp", 0)
    data.setdefault("expToNext", 100)
    data.setdefault("unspentAttributePoints", 0)
    data.setdefault("classId", None)
    data.setdefault("classSkills", [])
    attrs = data.get("attributes", {})
    for k in ("STR", "DEX", "CON", "INT", "WIS", "CHA"):
        if k in attrs:
            try:
                attrs[k] = max(1, min(20, int(attrs[k])))
            except (TypeError, ValueError):
                attrs[k] = 10
    return data
```

```python
# server/services/exp_formula.py
"""Mirror of client/src/services/level/expFormula.ts — server-authoritative."""
MAX_LEVEL = 20

_DIFFICULTY_MULT = {
    'easy': 0.5,
    'normal': 1.0,
    'hard': 1.5,
    'deadly': 2.0,
}


def exp_to_next(level: int) -> int:
    if level >= MAX_LEVEL:
        return 0
    return round(100 * (level ** 1.5))


def apply_exp_formula(level: int, exp: int, amount: int,
                      difficulty: str = 'normal') -> tuple[int, int]:
    mult = _DIFFICULTY_MULT.get(difficulty, 1.0)
    final = int(amount * mult)
    if level >= MAX_LEVEL or final <= 0:
        return (level, exp)

    pool = exp + final
    while level < MAX_LEVEL:
        need = exp_to_next(level)
        if pool < need:
            break
        pool -= need
        level += 1
    if level >= MAX_LEVEL:
        pool = 0
    return (level, pool)
```

```python
from pydantic import BaseModel
from ..services.exp_formula import apply_exp_formula, exp_to_next, MAX_LEVEL


class ExpGrantRequest(BaseModel):
    amount: int
    difficulty: str = 'normal'


class AttributeSpendRequest(BaseModel):
    attribute: str  # one of STR/DEX/CON/INT/WIS/CHA


@router.patch("/{char_id}/exp")
async def grant_exp(char_id: str, body: ExpGrantRequest,
                    repo=Depends(get_character_repo),
                    player_id=Depends(get_current_player)):
    char = await repo.get(char_id)
    if not char:
        raise HTTPException(404, "Character not found")
    if char.get("playerId") != player_id:
        raise HTTPException(403, "Not your character")

    new_level, new_exp = apply_exp_formula(
        char["level"], char["exp"], body.amount, body.difficulty
    )
    attr_points = char.get("unspentAttributePoints", 0) + max(0, new_level - char["level"])
    patch = {
        "level": new_level,
        "exp": new_exp,
        "expToNext": exp_to_next(new_level),
        "unspentAttributePoints": attr_points,
    }
    await repo.update(char_id, {**char, **patch})
    return patch


@router.patch("/{char_id}/attributes/spend")
async def spend_attribute_point(char_id: str, body: AttributeSpendRequest,
                                 repo=Depends(get_character_repo),
                                 player_id=Depends(get_current_player)):
    char = await repo.get(char_id)
    if not char:
        raise HTTPException(404, "Character not found")
    if char.get("playerId") != player_id:
        raise HTTPException(403, "Not your character")
    if char.get("unspentAttributePoints", 0) <= 0:
        raise HTTPException(400, "no unspent points")
    if body.attribute not in ("STR", "DEX", "CON", "INT", "WIS", "CHA"):
        raise HTTPException(400, "invalid attribute")
    attrs = dict(char.get("attributes", {}))
    new_val = min(20, attrs.get(body.attribute, 10) + 1)
    attrs[body.attribute] = new_val
    await repo.update(char_id, {
        **char,
        "attributes": attrs,
        "unspentAttributePoints": char["unspentAttributePoints"] - 1,
    })
    return {
        "attributes": attrs,
        "unspentAttributePoints": char["unspentAttributePoints"] - 1,
    }
```

```typescript
// client/src/services/level/grantExp.ts
import type { Character } from '../../types/character';
import { expToNext, MAX_LEVEL } from './expFormula';
import type { Difficulty } from '../../types/combat';  // adjust to existing Difficulty type

const DIFFICULTY_MULT: Record<Difficulty, number> = {
  easy: 0.5, normal: 1.0, hard: 1.5, deadly: 2.0,
};

export interface GrantExpResult {
  character: Character;
  leveledUpTo: number[];
  totalAwarded: number;
}

export function grantExp(character: Character, amount: number,
                         difficulty: Difficulty): GrantExpResult {
  if (character.level >= MAX_LEVEL || amount <= 0) {
    return { character, leveledUpTo: [], totalAwarded: 0 };
  }
  const mult = DIFFICULTY_MULT[difficulty] ?? 1.0;
  const final = Math.floor(amount * mult);
  let { level, exp } = character;
  let pool = exp + final;
  let consumed = 0;
  const leveledUpTo: number[] = [];
  while (level < MAX_LEVEL) {
    const need = expToNext(level);
    if (pool < need) break;
    pool -= need;
    consumed += need;
    level += 1;
    leveledUpTo.push(level);
  }
  if (level >= MAX_LEVEL) pool = 0;
  const unspentAttributePoints =
    character.unspentAttributePoints + leveledUpTo.length;
  return {
    character: { ...character, level, exp: pool, unspentAttributePoints },
    leveledUpTo,
    totalAwarded: consumed,
  };
}
```

```typescript
// client/src/services/class/classEffects.ts
import type { Character, Attributes } from '../../types/character';
import { getClass } from '../../data/classes';
import type { ClassNodeEffect, ClassNode } from '../../types/class';

export interface ClassBonus {
  attributeMods: Partial<Attributes>;
  hpMaxBonus: number;
  mpMaxBonus: number;
  dodgeThresholdBonus: number;
  damageModifier: number;
  expBonus: number;
  qteToleranceMs: number;
}

export const EMPTY_CLASS_BONUS: ClassBonus = {
  attributeMods: {}, hpMaxBonus: 0, mpMaxBonus: 0,
  dodgeThresholdBonus: 0, damageModifier: 0, expBonus: 0, qteToleranceMs: 0,
};

function applyEffect(bonus: ClassBonus, eff: ClassNodeEffect): void {
  switch (eff.type) {
    case 'attribute_mod':
      bonus.attributeMods[eff.attribute] = (bonus.attributeMods[eff.attribute] ?? 0) + eff.bonus;
      break;
    case 'hp_max_bonus': bonus.hpMaxBonus += eff.bonus; break;
    case 'mp_max_bonus': bonus.mpMaxBonus += eff.bonus; break;
    case 'dodge_threshold_bonus': bonus.dodgeThresholdBonus += eff.bonus; break;
    case 'damage_modifier': bonus.damageModifier += eff.bonus; break;
    case 'exp_bonus': bonus.expBonus += eff.bonus; break;
    case 'qte_tolerance': bonus.qteToleranceMs += eff.bonus; break;
  }
}

export function aggregateClassEffects(character: Character): ClassBonus {
  if (!character.classId) return { ...EMPTY_CLASS_BONUS };
  const def = getClass(character.classId);
  if (!def) return { ...EMPTY_CLASS_BONUS };
  const bonus: ClassBonus = {
    attributeMods: {}, hpMaxBonus: 0, mpMaxBonus: 0,
    dodgeThresholdBonus: 0, damageModifier: 0, expBonus: 0, qteToleranceMs: 0,
  };
  for (const node of character.classSkills) {
    const def2 = def.nodes.find((n: ClassNode) => n.id === node.nodeId);
    if (!def2) continue;
    applyEffect(bonus, def2.effect);
  }
  return bonus;
}
```

```typescript
const TIER_UNLOCK: Record<2 | 3 | 4, number> = { 2: 5, 3: 10, 4: 15 };

export function pendingTierChoice(character: Character): 2 | 3 | 4 | null {
  if (!character.classId) return null;
  const def = getClass(character.classId);
  if (!def) return null;
  for (const tier of [2, 3, 4] as const) {
    if (character.level >= TIER_UNLOCK[tier]) {
      const has = character.classSkills.some(n =>
        def.nodes.find(d => d.id === n.nodeId)?.tier === tier
      );
      if (!has) return tier;
    }
  }
  return null;
}

export function isValidClassNodeId(classId: string, nodeId: string): boolean {
  const def = getClass(classId);
  if (!def) return false;
  return def.nodes.some(n => n.id === nodeId);
```

```python
import re
from fastapi import HTTPException

ALLOWED_CLASSES = {'warrior', 'cleric', 'mage', 'thief'}
NODE_ID_RE = re.compile(r'^(warrior|cleric|mage|thief)_t[1-4]_[1-3]$')


def _expected_node_count(level: int) -> int:
    if level < 5: return 1
    if level < 10: return 2
    if level < 15: return 3
    return 4


def validate_class_update(char: dict, new_class_id: str | None,
                          new_class_skills: list[dict]) -> None:
    if new_class_id is not None and new_class_id not in ALLOWED_CLASSES:
        raise HTTPException(422, f"invalid classId: {new_class_id}")

    expected = _expected_node_count(char.get("level", 1))
    if len(new_class_skills) > expected:
        raise HTTPException(422,
            f"too many class skills ({len(new_class_skills)} > {expected})")

    for node in new_class_skills:
        nid = node.get("nodeId", "")
        if not NODE_ID_RE.match(nid):
            raise HTTPException(422, f"invalid nodeId format: {nid}")
        if node.get("classId") != new_class_id:
            raise HTTPException(422, "node classId mismatch")

    current = char.get("classId")
    if current is not None and current != new_class_id:
        raise HTTPException(422, "classId is locked in v0.5 (cannot change class)")

    if new_class_id is not None and len(new_class_skills) == 0:
        raise HTTPException(422, "must have at least T1 node when choosing class")
```

```tsx
interface Props {
  initialClassId: string | null;
  initialClassSkills: ClassSkillNode[];
  onChange: (classId: string | null, classSkills: ClassSkillNode[]) => void;
}

export function ClassStep({ initialClassId, initialClassSkills, onChange }: Props) {
  const [selectedClass, setSelectedClass] = useState<ClassId | null>(
    (initialClassId as ClassId) ?? null
  );
  const [selectedNode, setSelectedNode] = useState<string | null>(
    initialClassSkills[0]?.nodeId ?? null
  );
  const [pickingNode, setPickingNode] = useState(false);

  const handleClassPick = (cls: ClassId | null) => {
    setSelectedClass(cls);
    setSelectedNode(null);
    setPickingNode(cls !== null);
    if (cls === null) {
      onChange(null, []);
    }
  };

  const handleNodePick = (node: ClassNode) => {
    setSelectedNode(node.id);
    setPickingNode(false);
    onChange(selectedClass, [{
      classId: selectedClass!,
      nodeId: node.id,
      unlockedAt: Date.now(),  // or worldDay if accessible
    }]);
  };

  if (pickingNode && selectedClass) {
    const def = getClass(selectedClass);
    const t1Nodes = def?.nodes.filter(n => n.tier === 1) ?? [];
    return (
      <div data-testid="classstep-tier1">
        <h3>选择你的 T1 专精</h3>
        {t1Nodes.map(node => (
          <button key={node.id} data-testid={`classstep-node-${node.id}`} onClick={() => handleNodePick(node)}>
            <div>{node.name}</div>
            <div>{node.description}</div>
          </button>
        ))}
      </div>
    );
  }

  return (
    <div data-testid="classstep-classes">
      <h3>选择你的职业</h3>
      <button data-testid="classstep-none" onClick={() => handleClassPick(null)}>无职业</button>
      {CLASS_LIST.map(cls => (
        <button key={cls.id} data-testid={`classstep-class-${cls.id}`} onClick={() => handleClassPick(cls.id)}>
          <span>{cls.icon} {cls.name}</span>
          <span>{cls.description}</span>
        </button>
      ))}
    </div>
  );
}
```
